\section{Design Requirements and Formulas}

A percent error of less than $5\%$ is always required.

\subsection{Formulas for an Active High-Pass Filter}

\begin{align}
f_c &= \frac{1}{{2\pi R_1 C_1}} \\[10pt]
\end{align}

\begin{align}
A_{max} &= \frac{R_2}{{R_1}} \\[10pt]
\end{align}

where:
\begin{itemize}
    \item $f_c$ is the cutoff frequency
    \item $A_{max}$ is the maximum voltage gain of the high-pass filter
    \item $R_1$, $R_2$, are the resistor values
    \item $C_1$ is the capacitor value
    %as seen in figure
\end{itemize}

\subsection{Formulas for an Active Low-Pass Filter}

\begin{align}
f_c &= \frac{1}{{2\pi R_2 C_1}} \\[10pt]
\end{align}

\begin{align}
A_{max} &= \frac{R_2}{{R_1}} \\[10pt]
\end{align}

where:
\begin{itemize}
    \item $f_c$ is the cutoff frequency
    \item $A_{max}$ is the maximum voltage gain of the high-pass filter
    \item $R_1$, $R_2$, are the resistor values
    \item $C_1$ is the capacitor value
    %as seen in figure
\end{itemize}
